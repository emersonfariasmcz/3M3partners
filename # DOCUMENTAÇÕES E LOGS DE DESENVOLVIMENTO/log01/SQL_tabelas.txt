-- Tabela de Produtos
CREATE TABLE IF NOT EXISTS produtos (
    produto_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    produto_nome VARCHAR(100) NOT NULL,
    produto_descricao TEXT,
    produto_codigo_barras VARCHAR(50) UNIQUE,
    produto_unidade_medida_id INT, 
    produto_estoque_minimo INT DEFAULT 0,
    produto_estoque_maximo INT DEFAULT 0,
    produto_preco_custo DECIMAL(10,2) DEFAULT 0.00,
    produto_preco_venda DECIMAL(10,2) DEFAULT 0.00,
    categoria_id INT NOT NULL,
    fabricante_id INT,
    fornecedor_id INT,
    produto_status ENUM('ativo', 'inativo') DEFAULT 'ativo',
    produto_criadoem DATETIME DEFAULT CURRENT_TIMESTAMP,
    produto_atualizadoem DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    -- Chaves estrangeiras
    FOREIGN KEY (categoria_id) REFERENCES categorias(categoria_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (fabricante_id) REFERENCES fabricantes(fabricante_id) ON DELETE SET NULL ON UPDATE CASCADE,
    FOREIGN KEY (fornecedor_id) REFERENCES fornecedores(fornecedor_id) ON DELETE SET NULL ON UPDATE CASCADE,
    FOREIGN KEY (produto_unidade_medida_id) REFERENCES unidademedidas(unidademedida_id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Produtos do estoque';

    movimentacao_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    movimentacao_data_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    movimentacao_tipo_id INT NOT NULL,
    produto_id INT NOT NULL,
    movimentacao_quantidade DECIMAL(15, 3) NOT NULL, -- Permite decimais (ex: 1.5 Kg)
    movimentacao_unidade_medida VARCHAR(20) NOT NULL, -- Cópia da unidade do produto
    movimentacao_valor_unitario DECIMAL(10, 2) DEFAULT 0.00, -- Valor unitário na moeda local

    movimentacao_documento_referencia VARCHAR(100), -- Número da NF, Requisição, etc.

    movimentacao_usuario_id INT NOT NULL, -- FK para usuarios

    movimentacao_criada_em DATETIME DEFAULT CURRENT_TIMESTAMP,
